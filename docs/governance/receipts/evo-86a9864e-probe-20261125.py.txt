def run(fn, cases, label):
    bad = 0
    gaps = 0
    print("== " + label + " ==")
    for cid, cmd, expected in cases:
        actual = fn(cmd)
        if actual == expected:
            mark = "OK        "
        elif cid in KNOWN_GAPS:
            mark = "KNOWN_GAP "
            gaps += 1
        else:
            mark = "MISMATCH  "
            bad += 1
        print(mark + cid + " | expected=" + repr(expected) + " actual=" + repr(actual))
    return bad, gaps
