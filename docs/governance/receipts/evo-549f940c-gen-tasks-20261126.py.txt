def run(args, cwd):
    return subprocess.run(args, cwd=cwd, check=True, capture_output=True, text=True).stdout


def build(name, base_files, head_files):
    repo = D / name
    if repo.exists():
        shutil.rmtree(repo)
    repo.mkdir(parents=True)
    run(GIT + ["init", "-q"], repo)
    for rel, text in base_files.items():
        (repo / rel).write_text(text, encoding="utf-8")
    run(GIT + ["add", "-A"], repo)
    run(GIT + ["commit", "-q", "-m", "base"], repo)
    base = run(["git", "rev-parse", "HEAD"], repo).strip()
    for rel, text in head_files.items():
        (repo / rel).write_text(text, encoding="utf-8")
    run(GIT + ["add", "-A"], repo)
    run(GIT + ["commit", "-q", "-m", "candidate"], repo)
    head = run(["git", "rev-parse", "HEAD"], repo).strip()
    changed = run(["git", "diff", "--name-only", base, head], repo)
    diff = run(["git", "diff", base, head], repo)
    return repo, base, head, changed, diff


def template(src_path):
    tree = ast.parse(pathlib.Path(src_path).read_text(encoding="utf-8"))
    for node in ast.walk(tree):
        if isinstance(node, ast.FunctionDef) and node.name == "start_review":
            for stmt in node.body:
                target = stmt.targets[0] if isinstance(stmt, ast.Assign) else None
                if isinstance(target, ast.Name) and target.id == "task":
                    return compile(ast.Expression(stmt.value), str(src_path), "eval")
    raise SystemExit("task template not found in " + str(src_path))


def render(code, lane_id, base, head, worktree, changed, diff):
    return eval(code, {"json": json, "hashlib": hashlib, "lane_id": lane_id,
                       "base": base, "head": head, "lane": {"worktree": worktree},
                       "changed": changed, "diff": diff})


def main():
    arms = {
        "A": template(D / "host_verify_issuer_main.py"),
        "B": template(W / ".worktrees/evo-549f940c-merge-review-findings-20261126"
                      / "src/llm_loop/introspection/host_verify_issuer.py"),
    }
    candidates = {
        "revise": build("cand-revise", {"auth.py": AUTH_BASE, "test_auth.py": TESTS_BASE},
                        {"auth.py": AUTH_HEAD, "test_auth.py": TESTS_HEAD}),
        "pass": build("cand-pass", {"mathx.py": MATH_BASE, "test_mathx.py": MATH_TEST_BASE},
                      {"mathx.py": MATH_HEAD, "test_mathx.py": MATH_TEST_HEAD}),
    }
    meta = {}
    for cname, (repo, base, head, changed, diff) in candidates.items():
        lane_id = "lane-probe-549f940c-" + cname
        for arm, code in arms.items():
            task = render(code, lane_id, base, head, str(repo), changed, diff)
            out = D / f"task_{arm}_{cname}.txt"
            out.write_text(task, encoding="utf-8")
            meta[f"{arm}_{cname}"] = {
                "task_file": str(out), "task_chars": len(task), "lane_id": lane_id,
                "base": base, "head": head, "candidate_worktree": str(repo),
                "task_sha256": hashlib.sha256(task.encode()).hexdigest(),
            }
    (D / "candidates.json").write_text(json.dumps(meta, indent=2), encoding="utf-8")
    print(json.dumps({k: {kk: v[kk] for kk in ("task_chars", "head")} for k, v in meta.items()}, indent=1))


if __name__ == "__main__":
    main()
